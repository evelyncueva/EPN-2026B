\begin{objetivos}
\begin{itemize}
  \item Interpretar límites infinitos y límites al infinito mediante expresiones y gráficas.
  \item Identificar asíntotas verticales y horizontales a partir de límites.
  \item Calcular límites al infinito de funciones racionales mediante comparación de grados.
  \item Determinar el crecimiento no acotado de polinomios y funciones racionales cuando $x\to\pm\infty$.

\end{itemize}

\begin{resumen}
Un límite infinito describe valores de la función que crecen o decrecen sin cota cuando $x$ se aproxima a un número. Un límite al infinito describe el comportamiento de la función cuando $x$ toma valores positivos o negativos cada vez mayores en valor absoluto. Estas dos ideas permiten identificar asíntotas verticales y horizontales.
\end{resumen}

\section*{Funciones acotadas}

\begin{definicion}[función acotada]
Sea $f$ una función real con dominio no vacío $\operatorname{Dom}(f)$. Decimos que $f$ está \textbf{acotada superiormente} si existe un número real $M$ tal que
\[
  f(x)\leq M\qquad\text{para todo }x\in\operatorname{Dom}(f).
\]
El número $M$ se llama \textbf{cota superior}. Análogamente, $f$ está \textbf{acotada inferiormente} si existe un número real $m$ tal que
\[
  m\leq f(x)\qquad\text{para todo }x\in\operatorname{Dom}(f).
\]
El número $m$ se llama \textbf{cota inferior}.

La función está \textbf{acotada} si está acotada tanto superior como inferiormente; es decir, si existen $m,M\in\mathbb{R}$ tales que
\[
  m\leq f(x)\leq M\qquad\text{para todo }x\in\operatorname{Dom}(f).
\]
Equivalentemente, existe un número real $K>0$ tal que $|f(x)|\leq K$ para todo $x\in\operatorname{Dom}(f)$. Estas desigualdades deben cumplirse en todo el dominio de la función.
\end{definicion}

\begin{ejemplo}[función acotada]
La función $f(x)=\sin x$, con dominio $\mathbb{R}$, es acotada porque
\[
  -1\leq\sin x\leq1\qquad\text{para todo }x\in\mathbb{R}.
\]
Por tanto, $-1$ es una cota inferior y $1$ es una cota superior.
\end{ejemplo}

\begin{ejemplo}[función que no es acotada]
La función $g(x)=\dfrac{1}{x}$, con dominio $\mathbb{R}\setminus\{0\}$, no está acotada superiormente ni inferiormente. Al tomar valores positivos de $x$ cada vez más próximos a $0$, sus valores crecen sin cota; al tomar valores negativos de $x$ cada vez más próximos a $0$, sus valores decrecen sin cota. Por tanto, no es una función acotada.
\end{ejemplo}

\section*{Dos comportamientos diferentes}

Las expresiones
\[
  \lim_{x\to a}f(x)=+\infty
  \qquad\text{y}\qquad
  \lim_{x\to+\infty}f(x)=L
\]
responden preguntas distintas. En la primera, $x$ se aproxima al número $a$ y los valores de $f(x)$ crecen sin cota. En la segunda, $x$ crece sin cota y los valores de $f(x)$ se aproximan al número real $L$.

\begin{nota}
Los símbolos $+\infty$ y $-\infty$ no representan números reales. Por ello, un límite infinito describe un comportamiento no acotado; no afirma que la función alcance el valor infinito.
\end{nota}

\section*{Límites infinitos y asíntotas verticales}

\begin{definicion}[límite infinito]
Escribimos
\[
  \lim_{x\to a}f(x)=+\infty
\]
cuando los valores de $f(x)$ pueden hacerse positivos y tan grandes como se desee al tomar puntos $x$ del dominio suficientemente próximos a $a$, con $x\neq a$. De manera análoga,
\[
  \lim_{x\to a}f(x)=-\infty
\]
significa que los valores de $f(x)$ pueden hacerse negativos y tan grandes en valor absoluto como se desee.
\end{definicion}

Aunque en la definición anterior se escribe el límite bilateral $x\to a$, también se puede hablar de un límite infinito cuando este comportamiento ocurre por un solo lado, es decir, cuando alguno de los límites laterales es $+\infty$ o $-\infty$.

\begin{definicion}[asíntota vertical]
La recta $x=a$ es una \textbf{asíntota vertical} de la gráfica de $f$ si al menos uno de los límites laterales de $f$ en $a$ es $+\infty$ o $-\infty$.
\end{definicion}

Para que exista una asíntota vertical, basta con uno de estos límites laterales; no se requiere que exista el límite bilateral.

\begin{ejemplo}[signos diferentes a cada lado]
Sea
\[
  f\colon\mathbb{R}\setminus\{0\}\longrightarrow\mathbb{R},
  \qquad
  f(x)=\frac{1}{x}.
\]
Determine el comportamiento de $f$ cerca de $0$.
\end{ejemplo}
\ifmostrarSoluciones
\begin{solucion}
Cuando $x<0$, el número $x$ es negativo y se aproxima a $0$; por tanto, su recíproco decrece sin cota. Cuando $x>0$, el número $x$ es positivo y se aproxima a $0$; por tanto, su recíproco crece sin cota. Así,
\[
  \lim_{x\to0^-}\frac{1}{x}=-\infty,
  \qquad
  \lim_{x\to0^+}\frac{1}{x}=+\infty.
\]
La recta $x=0$ es una asíntota vertical. Como los comportamientos laterales tienen signos diferentes, no se escribe un único límite bilateral igual a $+\infty$ o a $-\infty$.
\end{solucion}
\else\vspace{20mm}\fi

\begin{ejemplo}[el mismo signo a ambos lados]
Calcule los límites laterales de
\[
  g(x)=\frac{1}{(x+1)^2},\qquad
  \operatorname{Dom}(g)=\mathbb{R}\setminus\{-1\}.
\]
cuando $x$ se aproxima a $-1$ e identifique la asíntota vertical.
\end{ejemplo}
\ifmostrarSoluciones
\begin{solucion}
Para $x\neq-1$, el denominador $(x+1)^2$ es positivo. Al aproximarse $x$ a $-1$ por cualquiera de los dos lados, el denominador se aproxima a $0$ mediante valores positivos. En consecuencia,
\[
  \lim_{x\to-1^-}\frac{1}{(x+1)^2}
  =\lim_{x\to-1^+}\frac{1}{(x+1)^2}
  =+\infty.
\]
En este caso también podemos escribir
\[
  \lim_{x\to-1}\frac{1}{(x+1)^2}=+\infty.
\]
La recta $x=-1$ es una asíntota vertical.
\end{solucion}
\else\vspace{18mm}\fi

\begin{nota}[un denominador nulo no es suficiente]
Que el denominador se anule en $x=a$ no garantiza una asíntota vertical. Por ejemplo,
\[
  \frac{x^2-4}{x-2}=x+2
  \qquad (x\neq2).
\]
Por tanto,
\[
  \lim_{x\to2}\frac{x^2-4}{x-2}=4,
\]
y la recta $x=2$ no es una asíntota vertical. Antes de decidir, se deben simplificar los factores comunes y estudiar el límite.
\end{nota}

\section*{Límites al infinito y asíntotas horizontales}

\begin{definicion}[límite al infinito]
Escribimos
\[
  \lim_{x\to+\infty}f(x)=L
\]
cuando los valores de $f(x)$ se aproximan a $L$ al tomar valores positivos de $x$ cada vez mayores. La expresión
\[
  \lim_{x\to-\infty}f(x)=L
\]
describe el comportamiento de $f(x)$ cuando $x$ toma valores negativos cuyo valor absoluto es cada vez mayor.
\end{definicion}

\begin{definicion}[asíntota horizontal]
La recta $y=L$ es una \textbf{asíntota horizontal} de la gráfica de $f$ si se cumple al menos una de las igualdades
\[
  \lim_{x\to+\infty}f(x)=L
  \qquad\text{o}\qquad
  \lim_{x\to-\infty}f(x)=L.
\]
\end{definicion}


\begin{ejemplo}[límites al infinito de la función recíproca]
Sea $f(x)=\dfrac{1}{x}$, con dominio $\mathbb{R}\setminus\{0\}$. Calcule sus límites cuando $x\to+\infty$ y cuando $x\to-\infty$, e identifique su asíntota horizontal.
\end{ejemplo}
\ifmostrarSoluciones
\begin{solucion}
Cuando $x$ crece en valor absoluto, $1/x$ se aproxima a $0$: toma valores positivos si $x>0$ y negativos si $x<0$. Por tanto,
\[
  \lim_{x\to+\infty}\frac{1}{x}=0,
  \qquad
  \lim_{x\to-\infty}\frac{1}{x}=0.
\]
La recta $y=0$ es una asíntota horizontal.
\end{solucion}
\else\vspace{18mm}\fi

\begin{ejemplo}[una asíntota horizontal]
Sea
\[
  h\colon\mathbb{R}\setminus\{-1\}\longrightarrow\mathbb{R},
  \qquad
  h(x)=\frac{2x+3}{x+1}.
\]
Determine su comportamiento cuando $x\to+\infty$ y cuando $x\to-\infty$.
\end{ejemplo}
\ifmostrarSoluciones
\begin{solucion}
Dividimos el numerador y el denominador entre $x$:
\[
  \frac{2x+3}{x+1}
  =\frac{2+3/x}{1+1/x}.
\]
Como
\[
  \lim_{x\to+\infty}\frac{1}{x}
  =\lim_{x\to-\infty}\frac{1}{x}
  =0,
\]
obtenemos
\[
  \lim_{x\to+\infty}\frac{2x+3}{x+1}
  =\lim_{x\to-\infty}\frac{2x+3}{x+1}
  =2.
\]
La recta $y=2$ es una asíntota horizontal.
\end{solucion}
\else\vspace{18mm}\fi

\section*{Comparación de grados}

Considere una función racional
\[
  r(x)=\frac{a_nx^n+\cdots+a_1x+a_0}
             {b_mx^m+\cdots+b_1x+b_0},
  \qquad
  a_n\neq0,
  \quad
  b_m\neq0.
\]
Su dominio es $\operatorname{Dom}(r)=\{x\in\mathbb{R}:b_mx^m+\cdots+b_1x+b_0\neq0\}$. Los términos de mayor grado determinan el comportamiento cuando $x\to+\infty$ o $x\to-\infty$.

\begin{proposicion}[límites al infinito de funciones racionales]
Para la función racional anterior se tienen los siguientes casos:
\begin{enumerate}
  \item Si $n<m$, entonces
  \[
    \lim_{x\to\pm\infty}r(x)=0.
  \]
  La recta $y=0$ es una asíntota horizontal.
  \item Si $n=m$, entonces
  \[
    \lim_{x\to\pm\infty}r(x)=\frac{a_n}{b_m}.
  \]
  La recta $y=a_n/b_m$ es una asíntota horizontal.
  \item Si $n>m$, estudiar el límite de $r(x)$ cuando $x\to+\infty$ o $x\to-\infty$ equivale a estudiar el límite de $cx^k$, donde
  \[
    c=\frac{a_n}{b_m},\qquad k=n-m>0.
  \]
  Es decir,
  \[
    \lim_{x\to+\infty}r(x)=\lim_{x\to+\infty}cx^k,
    \qquad
    \lim_{x\to-\infty}r(x)=\lim_{x\to-\infty}cx^k.
  \]
  Para calcular estos límites, usamos que, para todo entero positivo $k$,
  \[
    \lim_{x\to+\infty}x^k=+\infty,
  \]
  mientras que
  \[
    \lim_{x\to-\infty}x^k=
    \begin{cases}
      +\infty, & k\text{ es par},\\
      -\infty, & k\text{ es impar}.
    \end{cases}
  \]
  Al multiplicar por $c$, el signo del infinito se conserva si $c>0$ y se invierte si $c<0$. En este caso, la función no tiene asíntota horizontal.
\end{enumerate}
\end{proposicion}

\begin{ejemplo}[tres comparaciones de grados]
Calcule los límites al infinito de las siguientes funciones e identifique sus asíntotas horizontales:
\[
  f(x)=\frac{2x+1}{5x^2+4},
  \qquad
  g(x)=\frac{3x^2-x+1}{2x^2+5},
  \qquad
  q(x)=\frac{6x^3+1}{2x^2+1}.
\]
Las tres funciones tienen dominio $\mathbb{R}$, pues sus denominadores son positivos para todo número real.
\end{ejemplo}
\ifmostrarSoluciones
\begin{solucion}
En $f$, estudiar el límite equivale a estudiar el cociente de sus términos de mayor grado:
\[
  \lim_{x\to\pm\infty}f(x)
  =\lim_{x\to\pm\infty}\frac{2x}{5x^2}
  =\lim_{x\to\pm\infty}\frac{2}{5x}=0.
\]
Por tanto, $y=0$ es una asíntota horizontal. En $g$, el estudio del límite equivale al del cociente $3x^2/(2x^2)$:
\[
  \lim_{x\to\pm\infty}g(x)
  =\lim_{x\to\pm\infty}\frac{3x^2}{2x^2}
  =\frac{3}{2}.
\]
Así, $y=3/2$ es una asíntota horizontal. En $q$, tenemos la equivalencia asintótica
\[
  q(x)=\frac{6x^3+1}{2x^2+1}\sim\frac{6x^3}{2x^2}=3x
  \qquad (x\to\pm\infty).
\]
Por tanto,
\[
  \lim_{x\to+\infty}q(x)=\lim_{x\to+\infty}3x=+\infty,
\]
\[
  \lim_{x\to-\infty}q(x)=\lim_{x\to-\infty}3x=-\infty.
\]
Por tanto, $q$ no tiene asíntota horizontal.
\end{solucion}
\else\vspace{24mm}\fi

\section*{Crecimiento no acotado cuando $x\to\pm\infty$}

Como se evidenció en la proposición anterior, cuando $x\to+\infty$ o $x\to-\infty$, los valores de una función pueden tender a $+\infty$ o a $-\infty$. En estos casos, la función presenta un crecimiento o decrecimiento no acotado. El signo del límite se determina a partir del coeficiente y de la paridad de la potencia dominante, como se muestra a continuación.

\begin{ejemplo}[signo y paridad del término dominante]
Calcule los límites al infinito de
\[
  f(x)=\frac{3x^3-x}{x^2+1}
  \qquad\text{y}\qquad
  g(x)=\frac{-2x^4+x}{x^2+3}.
\]
Ambas funciones tienen dominio $\mathbb{R}$.
\end{ejemplo}
\ifmostrarSoluciones
\begin{solucion}
La función $f$ se comporta como $3x$. Como la potencia dominante es impar y su coeficiente es positivo,
\[
  \lim_{x\to+\infty}f(x)=+\infty,
  \qquad
  \lim_{x\to-\infty}f(x)=-\infty.
\]
La función $g$ se comporta como $-2x^2$. La potencia dominante es par, por lo que $x^2$ es positivo en ambos extremos; el coeficiente negativo determina el signo final:
\[
  \lim_{x\to+\infty}g(x)=-\infty,
  \qquad
  \lim_{x\to-\infty}g(x)=-\infty.
\]
Ninguna de las dos funciones tiene asíntota horizontal.
\end{solucion}

\section*{Caso especial: raíces con $x^2$}

La identidad correcta es
\[
  \sqrt{x^2}=|x|,
\]
no $\sqrt{x^2}=x$ para todo número real $x$. Esta diferencia es esencial cuando se estudian los dos extremos del eje real.

\begin{nota}[: propiedades y teoremas para límites al infinito]
Las propiedades de los límites estudiadas cuando $x\to a$, con $a$ finito, también se aplican cuando $x\to+\infty$ o $x\to-\infty$, siempre que se cumplan sus respectivas hipótesis. En particular, si los límites de las funciones son números reales, se conservan las reglas de suma, resta, producto y cociente (con límite del denominador distinto de cero), así como las de potencias y raíces en los casos en que estén definidas. Por ejemplo, si $u(x)\to L\geq0$ y $u(x)\geq0$ para los valores de $x$ considerados, entonces
\[
  \lim_{x\to\pm\infty}\sqrt{u(x)}=\sqrt{L}.
\]
Estas reglas no permiten operar con $+\infty$ y $-\infty$ como si fueran números reales.

También se extienden los teoremas de comparación, como el \textbf{teorema del sánduche}: si
\[
  g(x)\leq f(x)\leq h(x)
\]
para todo $x$ suficientemente grande (hacia $+\infty$), o suficientemente negativo (hacia $-\infty$), y los límites de $g$ y $h$ en ese mismo extremo son iguales a $L$, entonces el límite de $f$ también es $L$.
\end{nota}

\begin{ejemplo}[dos asíntotas horizontales]
Calcule
\[
  \lim_{x\to+\infty}\frac{\sqrt{x^2+1}}{x}
  \qquad\text{y}\qquad
  \lim_{x\to-\infty}\frac{\sqrt{x^2+1}}{x}.
\]
\end{ejemplo}
\ifmostrarSoluciones
\begin{solucion}
Factorizamos $x^2$ dentro de la raíz:
\[
  \frac{\sqrt{x^2+1}}{x}
  =\frac{\sqrt{x^2\left(1+1/x^2\right)}}{x}
  =\frac{|x|}{x}\sqrt{1+\frac{1}{x^2}}.
\]
Si $x\to+\infty$, entonces $|x|/x=1$; si $x\to-\infty$, entonces $|x|/x=-1$. Por tanto,
\[
  \lim_{x\to+\infty}\frac{\sqrt{x^2+1}}{x}=1,
  \qquad
  \lim_{x\to-\infty}\frac{\sqrt{x^2+1}}{x}=-1.
\]
La gráfica tiene dos asíntotas horizontales: $y=1$ hacia la derecha y $y=-1$ hacia la izquierda.
\end{solucion}

\begin{nota}[ideas que deben quedar claras]
\begin{itemize}
  \item Una asíntota vertical se decide mediante límites cerca de un número; una asíntota horizontal se decide mediante límites cuando $x\to\pm\infty$.
  \item Un cero del denominador no garantiza una asíntota vertical: primero se revisan factores comunes.
  \item La comparación de grados permite decidir rápidamente los límites al infinito de una función racional.
  \item Cuando domina una potencia, la paridad del exponente y el signo del coeficiente determinan el comportamiento hacia $+\infty$ y $-\infty$.
  \item En expresiones con raíces, debe recordarse que $\sqrt{x^2}=|x|$.
\end{itemize}
\end{nota}
