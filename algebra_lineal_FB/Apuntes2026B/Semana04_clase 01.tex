\section{Sistemas de ecuaciones}

En las clases anteriores resolvimos sistemas de la forma $Ax=b$ mediante
operaciones elementales por fila. Consideremos ahora una situación en la que
la matriz de coeficientes se mantiene, pero los datos (el lado derecho) cambian.

\setcounter{subsection}{2}
\subsection{Matriz inversa}


\begin{ejem}[Motivación a la función inversa]\label{sem04cl01:motivacion}
    Supongamos dos números reales $x_1$ y $x_2$. Si su suma es $b_1$ y su
    diferencia es $b_2$, el sistema correspondiente es
    \[
        \left\{\begin{aligned}
            x_1+x_2&=b_1,\\
            x_1-x_2&=b_2.
        \end{aligned}\right.
    \]
    En forma matricial, este sistema se escribe como
    \[
        \underbrace{\begin{pmatrix}1&1\\1&-1\end{pmatrix}}_{A}
        \begin{pmatrix}x_1\\x_2\end{pmatrix}
        =\underbrace{\begin{pmatrix}b_1\\b_2\end{pmatrix}}_{b}.
    \]
    Si $b=(10,2)^{\top}$, al resolver el sistema se obtiene
    $x=(6,4)^{\top}$. Si los datos cambian a $b=(14,6)^{\top}$, debemos
    resolver otro sistema con la misma matriz $A$ y obtenemos
    $x=(10,4)^{\top}$.

    Repetir la eliminación cada vez resulta poco práctico. Sería útil encontrar
    una matriz que dependiera solo de $A$ y que, al multiplicarla por cualquier
    columna $b$, produjera directamente la solución. Como veremos, si $A$ posee
    una matriz inversa $A^{-1}$, cada sistema se resolverá simplemente mediante
    \[
        x=A^{-1}b.
    \]
    A continuación definiremos esta matriz y aprenderemos a calcularla.
\end{ejem}

\setcounter{defi}{3}
\begin{defi}[Matriz inversa]\label{sem04cl01:def-inversa}
    Sea $A\in\Mat[\R]{n}{n}$. Se dice que $A$ es \textbf{invertible} o
    \textbf{no singular} si existe una matriz $B\in\Mat[\R]{n}{n}$ tal que
    \[
        AB=BA=I_n.
    \]
    En este caso, $B$ se llama \textbf{inversa} de $A$ y se denota por
    $A^{-1}$. Si tal matriz no existe, $A$ se dice \textbf{singular}.
\end{defi}


\begin{prop}[Unicidad de la inversa]
    Si una matriz tiene inversa, esta es única.
\end{prop}

\begin{teo}[Solución mediante la matriz inversa]\label{sem04cl01:teo-solucion-inversa}
    Sean $A\in\Mat[\R]{n}{n}$ una matriz invertible y
    $b\in\Mat[\R]{n}{1}$. El sistema $Ax=b$ tiene como única solución
    \[
        x=A^{-1}b.
    \]
\end{teo}

\begin{proof}
    Como $A^{-1}$ existe, podemos multiplicar ambos lados de $Ax=b$ por
    $A^{-1}$ a la izquierda:
    \begin{align*}
        Ax=b
        &\Longleftrightarrow A^{-1}(Ax)=A^{-1}b\\
        &\Longleftrightarrow (A^{-1}A)x=A^{-1}b\\
        &\Longleftrightarrow I_nx=A^{-1}b\\
        &\Longleftrightarrow x=A^{-1}b.
    \end{align*}
    Por tanto, $x=A^{-1}b$ satisface el sistema y es la única solución posible.
\end{proof}

\begin{ejem}[Aplicación del teorema]
    Si
    \[
        A^{-1}=\begin{pmatrix}1&-1\\0&1\end{pmatrix}
        \qquad\text{y}\qquad
        b=\begin{pmatrix}5\\2\end{pmatrix},
    \]
    entonces la solución de $Ax=b$ es
    \[
        x=A^{-1}b
        =\begin{pmatrix}1&-1\\0&1\end{pmatrix}
         \begin{pmatrix}5\\2\end{pmatrix}
        =\begin{pmatrix}3\\2\end{pmatrix}.
    \]
\end{ejem}

\subsubsection{Cálculo de la matriz inversa}

Las operaciones por fila que usamos para resolver sistemas también permiten
decidir si una matriz es invertible y, cuando lo es, calcular su inversa.

\begin{teo}[Cálculo de la matriz inversa]\label{sem04cl01:teo-calculo-inversa}
    Sea $A\in\Mat[\R]{n}{n}$. La matriz $A$ es invertible si y solo si es
    equivalente por filas a $I_n$. En tal caso,
    \[
        (A\mid I_n)\sim(I_n\mid A^{-1}).
    \]
\end{teo}

\noindent\textbf{Procedimiento.}
\begin{enumerate}
    \item Formar la matriz ampliada $(A\mid I_n)$.
    \item Aplicar Gauss--Jordan a toda la matriz ampliada.
    \item Si el bloque izquierdo se convierte en $I_n$, el bloque derecho es
          $A^{-1}$. Si el bloque izquierdo no puede convertirse en $I_n$, la
          matriz es singular.
\end{enumerate}

\begin{ejem}[Cálculo y comprobación]\label{sem04cl01:ej-inversa}
    Calculemos la inversa de la matriz del ejemplo
    \ref{sem04cl01:motivacion}:
    \[
        A=\begin{pmatrix}1&1\\1&-1\end{pmatrix}.
    \]
    Aplicamos las mismas operaciones por fila a ambos bloques:
    \begin{align*}
        \left(\begin{array}{cc|cc}1&1&1&0\\1&-1&0&1\end{array}\right)
        &\sim
        \left(\begin{array}{cc|cc}1&1&1&0\\0&-2&-1&1\end{array}\right)
        &&-F_1+F_2\to F_2\\
        &\sim
        \left(\begin{array}{cc|cc}1&1&1&0\\0&1&\frac12&-\frac12\end{array}\right)
        &&-\frac12F_2\to F_2\\
        &\sim
        \left(\begin{array}{cc|cc}1&0&\frac12&\frac12\\0&1&\frac12&-\frac12\end{array}\right)
        &&-F_2+F_1\to F_1.
    \end{align*}
    Por tanto,
    \[
        A^{-1}=\frac12\begin{pmatrix}1&1\\1&-1\end{pmatrix}.
    \]
    La comprobación $AA^{-1}=I_2$ confirma el cálculo.
\end{ejem}

Retomemos ahora el ejemplo \ref{sem04cl01:motivacion}. Una vez calculada la
inversa, no necesitamos repetir la eliminación cuando cambia el lado derecho:
\[
    x=A^{-1}b
    =\frac12\begin{pmatrix}1&1\\1&-1\end{pmatrix}
     \begin{pmatrix}b_1\\b_2\end{pmatrix}
    =\begin{pmatrix}(b_1+b_2)/2\\(b_1-b_2)/2\end{pmatrix}.
\]
Así, $b=(10,2)^{\top}$ produce $x=(6,4)^{\top}$ y
$b=(14,6)^{\top}$ produce $x=(10,4)^{\top}$ mediante una sola multiplicación.
La matriz inversa depende únicamente de $A$ y puede reutilizarse para cada
nuevo lado derecho $b$.

\begin{prop}[Propiedades de la inversa]\label{sem04cl01:prop-inversa}
    Sean $A,B\in\Mat[\R]{n}{n}$ matrices invertibles. Entonces:
    \begin{enumerate}
        \item $(A^{-1})^{-1}=A$;
        \item $AB$ es invertible y $(AB)^{-1}=B^{-1}A^{-1}$;
        \item $A^{\top}$ es invertible y
              $(A^{\top})^{-1}=(A^{-1})^{\top}$.
    \end{enumerate}
\end{prop}

\begin{proof}
    Cada afirmación se verifica comprobando que la matriz propuesta como
    inversa produce la identidad al multiplicarla tanto por la izquierda como
    por la derecha. Por ejemplo,
    \[
        (AB)(B^{-1}A^{-1})
        =A(BB^{-1})A^{-1}=I_n,
    \]
    y análogamente $(B^{-1}A^{-1})(AB)=I_n$.
\end{proof}

\begin{advertencia}
    En general, el orden importa: $(AB)^{-1}=B^{-1}A^{-1}$, no
    $A^{-1}B^{-1}$.
\end{advertencia}

\begin{ejem}[Despejar una matriz]
    Sean $A$ y $B$ matrices invertibles y supongamos que $APB=C$. Si deseamos
    despejar $P$, multiplicamos por $A^{-1}$ a la izquierda y por $B^{-1}$ a
    la derecha, respetando el orden, y obtenemos
    \[
        P=A^{-1}CB^{-1}.
    \]
\end{ejem}

\begin{ejem}[Cómo detectar que una matriz es singular]
    Para $C=\begin{pmatrix}1&2\\2&4\end{pmatrix}$ se obtiene
    \[
        \left(\begin{array}{cc|cc}1&2&1&0\\2&4&0&1\end{array}\right)
        \sim
        \left(\begin{array}{cc|cc}1&2&1&0\\0&0&-2&1\end{array}\right).
    \]
    Podemos verificar de varias formas que $C$ no tiene inversa:
    \begin{itemize}
        \item El bloque izquierdo tiene una fila nula y no puede reducirse a
              $I_2$.
        \item Se tiene $\rang(C)=1<2$, por lo que $C$ no tiene rango completo.
        \item Para el sistema homogéneo $Cx=0$, la matriz ampliada se escalona
              como
              \[
                  \left(\begin{array}{cc|c}1&2&0\\2&4&0\end{array}\right)
                  \sim
                  \left(\begin{array}{cc|c}1&2&0\\0&0&0\end{array}\right),
              \]
              Tendríamos infinitas soluciones no triviales. Pero si $C$ fuera invertible, multiplicar
              $Cx=0$ por $C^{-1}$ nos estregaría únicamente la solución trivial $x=0$.
    \end{itemize}
\end{ejem}

\begin{prop}[Criterios equivalentes de invertibilidad]
    Para $A\in\Mat[\R]{n}{n}$ son equivalentes las siguientes afirmaciones:
    \begin{enumerate}
        \item $A$ es invertible;
        \item $A$ es equivalente por filas a $I_n$;
        \item $\rang(A)=n$;
        \item $Ax=0$ tiene únicamente la solución trivial;
        \item $Ax=b$ tiene una única solución para cada
              $b\in\Mat[\R]{n}{1}$.
    \end{enumerate}
\end{prop}

Esta proposición reúne lo estudiado sobre rango, sistemas homogéneos y
soluciones de sistemas. Si $A$ es invertible, la solución única de $Ax=b$ es
\[
    x=A^{-1}b.
\]
Una matriz cuadrada es invertible exactamente cuando
su reducción por filas tiene un pivote en cada columna. En la siguiente clase
veremos que esta misma condición puede expresarse mediante un solo número, el determinante: $A$ es invertible si y solo si $\det(A)\ne0$.

\section{Ejercicios}

\begin{ejer}[Propiedades de la inversa]
    Sean $A$ y $X$ matrices invertibles. Despeje $X$ en
    $A^{\top}X^{-1}=B$ y verifique el resultado por sustitución.
\end{ejer}

\begin{ejer}[Aplicación: un mismo modelo, distintos datos]
    Un taller fabrica dos tipos de kits, $P$ y $Q$. Cada kit $P$ requiere dos
    horas de corte y una de ensamblaje; cada kit $Q$ requiere una hora de corte
    y una de ensamblaje. Las horas de corte y de ensamblaje programadas se
    utilizarán por completo.
    \begin{enumerate}
        \item Si $x_1$ y $x_2$ son las cantidades producidas y $b_1,b_2$ son,
              respectivamente, las horas programadas de corte y ensamblaje,
              plantee el sistema de ecuaciones y escríbalo en la forma
              $Ax=b$. Identifique las matrices $A$, $x$ y $b$.
        \item Calcule $A^{-1}$ mediante el teorema
              \ref{sem04cl01:teo-calculo-inversa}. Como comprobación de los
              dos pasos anteriores, verifique que
              \[
                  x=A^{-1}b
                  =\begin{pmatrix}b_1-b_2\\-b_1+2b_2\end{pmatrix}.
              \]
        \item Calcule la producción para $b=(11,7)^{\top}$,
              $b=(13,8)^{\top}$ y $b=(17,10)^{\top}$ sin repetir la
              eliminación de Gauss--Jordan.
        \item Explique qué indicaría una componente negativa de $A^{-1}b$.
    \end{enumerate}
\end{ejer}

\begin{ejer}[Práctica de Gauss--Jordan]
    Para cada matriz, determine si es invertible. Si lo es, calcule su inversa
    mediante el teorema \ref{sem04cl01:teo-calculo-inversa} y compruebe el
    resultado por multiplicación.
    \[
        A=\begin{pmatrix}1&2\\3&5\end{pmatrix},
        \qquad
        B=\begin{pmatrix}1&2&0\\0&1&2\\0&0&1\end{pmatrix},
        \qquad
        C=\begin{pmatrix}1&-1\\-2&2\end{pmatrix}.
    \]
\end{ejer}
