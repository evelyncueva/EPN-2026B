\noindent\textbf{Ejercicios}
\begin{enumerate}
  \item Para $a\in\mathbb{R}$, considere la función
  \[
    F_a\colon\mathbb{R}\to\mathbb{R},
    \qquad
    F_a(x)=
    \begin{cases}
      \dfrac{x^2+ax-2a-4}{|x-2|}, & x\neq2,\\[2mm]
      7, & x=2.
    \end{cases}
  \]
  Calcule los límites laterales de $F_a$ cuando $x$ tiende a $2$ y determine
  todos los valores de $a$ para los cuales existe
  \[
    \lim_{x\to2}F_a(x).
  \]
  \textbf{Hint:} Factorice el numerador y considere los casos $x<2$ y $x>2$ por separado.
  \item Sea $A$ una función definida para $x$ cercanos a $0$, excepto posiblemente en $0$, que satisface
  \[
    |A(x)|\leq3.
  \]
  Calcule mediante el teorema del Sanduche
  \[
    \lim_{x\to0}
    \frac{x^2}{|x|+x^2}\,A(x).
  \]
\textbf{Hint:} Note que $x^2=|x|^2$ para $x\neq 0$.
  \item Considere las funciones
  \[
    f\colon\mathbb{R}\setminus\{-1\}
    \to\mathbb{R}\setminus\{2\},
    \qquad
    f(x)=\frac{2x+1}{x+1},
  \]
  y
  \[
    g\colon\mathbb{R}\setminus\{2\}
    \to\mathbb{R}\setminus\{-1\},
    \qquad
    g(x)=\frac{1-x}{x-2}.
  \]
  \begin{enumerate}
    \item Determine si cada función es inyectiva, sobreyectiva o biyectiva.
    \item Determine la función $g\circ f$.
  \end{enumerate}
\end{enumerate}

\ifmostrarSoluciones
\begin{solucion}[de la propuesta paralela]
\begin{enumerate}
  \item El numerador se factoriza como
  \[
    x^2+ax-2a-4=(x-2)(x+a+2).
  \]
  Si $x<2$, entonces $|x-2|=-(x-2)$ y
  \[
    F_a(x)=-(x+a+2).
  \]
  Si $x>2$, entonces $|x-2|=x-2$ y
  \[
    F_a(x)=x+a+2.
  \]
  Por tanto,
  \[
    \lim_{x\to2^-}F_a(x)=-(a+4),
    \qquad
    \lim_{x\to2^+}F_a(x)=a+4.
  \]
  El límite bilateral existe si y solo si
  \[
    -(a+4)=a+4,
  \]
  de donde $a=-4$. En ese caso,
  \[
    \lim_{x\to2}F_{-4}(x)=0.
  \]

  \item Para $x\neq0$, se tiene $x^2=|x|^2$ y $|x|>0$. Por tanto,
  \[
    \frac{x^2}{|x|+x^2}
    =\frac{|x|^2}{|x|(1+|x|)}
    =\frac{|x|}{1+|x|}.
  \]
  Como este factor es no negativo y $|A(x)|\leq3$, obtenemos
  \[
    \left|
      \frac{x^2}{|x|+x^2}A(x)
    \right|
    \leq
    3\frac{x^2}{|x|+x^2}
    =3\frac{|x|}{1+|x|}.
  \]
  En consecuencia,
  \[
    -\frac{3|x|}{1+|x|}
    \leq
    \frac{x^2}{|x|+x^2}A(x)
    \leq
    \frac{3|x|}{1+|x|}.
  \]
  Como $|x|$ tiende a $0$ y $1+|x|$ tiende a $1$, ambas cotas tienden a $0$.
  El teorema del Sanduche permite concluir que
  \[
    \lim_{x\to0}
    \frac{x^2}{|x|+x^2}A(x)=0.
  \]

  \item Si $f(x)=2$, entonces $2x+1=2(x+1)$, lo cual implicaría $1=2$.
  Por tanto, $f(x)\neq2$. De manera análoga, si $g(x)=-1$, se obtendría
  $1-x=-(x-2)$, que también implicaría $1=2$. Las dos leyes producen valores
  en sus codominios.

  Para estudiar $f$, sea $y\in\mathbb{R}\setminus\{2\}$. Al resolver
  \[
    y=\frac{2x+1}{x+1}
  \]
  para $x$, se obtiene el único valor
  \[
    x=\frac{1-y}{y-2}.
  \]
  Este valor pertenece al dominio de $f$, pues
  \[
    x+1=-\frac{1}{y-2}\neq0.
  \]
  Al sustituirlo en la ley de $f$ se verifica que $f(x)=y$. Así, cada
  elemento del codominio tiene una única preimagen y $f$ es inyectiva y
  sobreyectiva; por tanto, es biyectiva.

  Para $g$, sea $y\in\mathbb{R}\setminus\{-1\}$. Resolviendo
  \[
    y=\frac{1-x}{x-2}
    \iff x(y+1)=2y+1,
  \]
  obtenemos la única preimagen posible
  \[
    x=\frac{2y+1}{y+1}.
  \]
  Este valor pertenece al dominio de $g$, pues
  \[
    x-2=-\frac{1}{y+1}\neq0.
  \]
  Al sustituirlo en $g$ se obtiene $g(x)=y$. Cada elemento del codominio tiene
  una única preimagen, de modo que $g$ también es inyectiva, sobreyectiva y
  biyectiva.

  Para todo $x\in\mathbb{R}\setminus\{-1\}$,
  \begin{align*}
    (g\circ f)(x)
    &=\frac{1-\dfrac{2x+1}{x+1}}
            {\dfrac{2x+1}{x+1}-2}\\
    &=\frac{-x/(x+1)}{-1/(x+1)}\\
    &=x.
  \end{align*}
  No hay restricciones adicionales: si $x\neq-1$, entonces $f(x)\neq2$, por
  lo que se puede aplicar $g$. Así, la función solicitada es
  \[
    g\circ f\colon\mathbb{R}\setminus\{-1\}
    \to\mathbb{R}\setminus\{-1\},
    \qquad
    (g\circ f)(x)=x.
  \]
  Es la función identidad en $\mathbb{R}\setminus\{-1\}$.
\end{enumerate}
\end{solucion}
